import ast
import os

PKGS = {"recog", "plan", "execution", "common"}


def pkg_of(p):
    d = os.path.dirname(p)
    return d.replace(os.sep, ".") if d else ""


def resolve(mod):
    p = os.path.join(*mod.split("."))
    if os.path.isfile(p + ".py"):
        return p + ".py"
    ini = os.path.join(p, "__init__.py")
    return ini if os.path.isfile(ini) else None


seen, stack = {"main.py"}, ["main.py"]
while stack:
    src = stack.pop()
    here = pkg_of(src)
    names = set()
    for n in ast.walk(ast.parse(open(src, encoding="utf-8").read())):
        if isinstance(n, ast.Import):
            names |= {a.name for a in n.names}
        elif isinstance(n, ast.ImportFrom):
            if n.level:
                parts = here.split(".") if here else []
                base = ".".join(parts[:len(parts) - (n.level - 1)] or parts)
                pre = base + ("." + n.module if n.module else "")
            elif n.module:
                pre = n.module
            else:
                continue
            names.add(pre)
            names |= {pre + "." + a.name for a in n.names}
    for mod in names:
        top = mod.split(".")[0]
        if top not in PKGS:
            continue
        ini = os.path.join(top, "__init__.py")
        if os.path.isfile(ini):
            seen.add(ini)
        f = resolve(mod) or resolve(".".join(mod.split(".")[:-1]) or mod)
        if f and f not in seen:
            seen.add(f)
            stack.append(f)

files = sorted(x.replace(os.sep, "/") for x in seen)
print(len(files))
for f in files:
    print(f)
